cmake_minimum_required(VERSION 3.19)
project(vision_acceptance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "Production source root")
set(ONNXRUNTIME_ROOT "" CACHE PATH "ONNX Runtime installation directory")
if(NOT EXISTS "${ONNXRUNTIME_ROOT}/include/onnxruntime_cxx_api.h")
    message(FATAL_ERROR "Set ONNXRUNTIME_ROOT to the existing ORT installation")
endif()
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/prepare.py ${SOURCE_ROOT} ${CMAKE_CURRENT_BINARY_DIR}/generated COMMAND_ERROR_IS_FATAL ANY)
add_executable(acceptance_benchmark ${CMAKE_CURRENT_BINARY_DIR}/generated/acceptance_benchmark.cpp)
add_executable(acceptance_verify verify.cpp)
add_executable(acceptance_verify_instrumented verify.cpp)
target_compile_definitions(acceptance_verify_instrumented PRIVATE ACCEPTANCE_INSTRUMENTED)
foreach(target acceptance_benchmark acceptance_verify acceptance_verify_instrumented)
    target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/generated ${SOURCE_ROOT} ${OpenCV_INCLUDE_DIRS} ${ONNXRUNTIME_ROOT}/include)
    target_link_directories(${target} PRIVATE ${ONNXRUNTIME_ROOT}/lib)
    target_link_libraries(${target} PRIVATE onnxruntime ${OpenCV_LIBS})
    target_compile_options(${target} PRIVATE -Wall -Wextra)
    set_target_properties(${target} PROPERTIES BUILD_RPATH "${ONNXRUNTIME_ROOT}/lib")
endforeach()
